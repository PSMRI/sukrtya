After creating EKS run below command one by one
===============================================

helm install my-release bitnami/nginx
kubectl get svc --namespace default -w my-release-nginx
 kubectl get svc --namespace default

 kubectl get svc my-release-nginx --namespace default
 kubectl get svc my-release-nginx --namespace default -o yaml
///==========for internal ip==============================================================
export SERVICE_PORT=$(kubectl get --namespace default -o jsonpath="{.spec.ports[0].port}" services my-release-nginx)
export SERVICE_IP=$(kubectl get svc --namespace default my-release-nginx -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
echo "http://${SERVICE_IP}:${SERVICE_PORT}"
///======for external Ip ======================================================
SERVICE_IP=$(kubectl get svc my-release-nginx --namespace default -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
SERVICE_PORT=$(kubectl get svc my-release-nginx --namespace default -o jsonpath='{.spec.ports[0].port}')
echo "http://${SERVICE_IP}:${SERVICE_PORT}"
